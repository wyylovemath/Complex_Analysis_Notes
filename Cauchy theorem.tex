\chapter{Cauchy's Theorem}

\section{Gourset's Theorem}

\noindent\textbf{Intuition:} Recall the Fundamental Thm of Calculus for Contour Integrals implies that: If $\gamma$ is a closed curve in an open set $\Omega \subseteq \mathbb{C}$ and $f$ is continuous and has a primitive in $\Omega$, then
\[
\int_\gamma f(z)dz= 0
\]
We naturally ask whether the requirements for $f$ can be simplified. We may do so by restricting $\gamma$.

\begin{theorem}{Gourset's Theorem}\\
Let $\Omega$ be an open subset of $\mathbb{C}$ and let $T \subseteq \Omega$ be a triangle with interior contained in $\Omega$. If $f$ is holomorphic in $\Omega$, then
\[
\int_T f(z)dz=0
\]
\end{theorem}

\begin{proof}
    We denote by $T^{(0)}$ the original triangle. Then we can get four smaller triangles by $T_1^{(1)},T_2^{(1)}, T_3^{(1)}, T_4^{(1)}$ by connecting the midpoints of the three sides of $T^{(0)}$. Then
    \[
    \int_Tf(z) dz= \int_{T^{(0)}} f(z)dz= \sum_{j=1}^4 \int_{T_j^{(1)}} f(z)dz
    \]
    Hence, $\exists j\in \{1,2,3,4\}$, s.t.: $|\int_T f(z)dz| \leq 4|\int_{T_j^{(1)}} f(z)dz|$. Denote $T^{(1)}:= T_j^{(1)}$ and define $\mathcal{T}^{(1)}$ to be the solid triangle with boundary $T^{(1)}$. Repeat this process, and then we can get a sequence of solid triangles $\mathcal{T}^{(0)} \supseteq \mathcal{T}^{(1)} \supseteq \cdots \supseteq \mathcal{T}^{(n)} \supseteq \cdots$, such that
    \[
    |\int_T f(z)dz| \leq 4^n|\int_{T^{(n)}} f(z)dz|
    \]
    Note that the diameters of $\{T^{(n)}\}_{n=0}^\infty$ converge to $0$. Hence, by the Nested Thm, $\{z_0 \} =\bigcap_{n=0}^\infty \mathcal{T}^{(n)}$ for some $z_0 \in \Omega$. Since $f$ is holomorphic at $z_0$, then we can expand $f$ at $z_0$ into a first-order approximation:
    \[
    f(z)= f(z_0)+ f'(z_0) (z-z_0)+ \psi(z) (z-z_0)
    \]
    where $\psi(z)$ denotes the higher-order infinitesimal, i.e.: $\psi(z) \to 0$ as $z\to z_0$. Since the constant function $f(z_0)$ and the linear function $f'(z_0) (z-z_0)$ have primitives, then by the Fundamental Thm of Calculus for Contour Integrals, we have:
    \[
    \int_{T^{(n)}} f(z)dz =\int_{T^{(n)}} \psi(z) (z-z_0)dz
    \]
    Let $p_n$ denote the perimeter of $T^{(n)}$ and $d_n$ denote the diameter of $T^{(n)}$. Then $p_n= 2^{-n} p_0$ and $d_n= 2^{-n}d_0$. Therefore, 
    \[
    |\int_Tf(z)dz| \leq 4^n|\int_{T^{(n)}} f(z)dz| =4^n|\int_{T^{(n)}} \psi(z) (z-z_0)dz| < p_0d_0 \epsilon
    \]
\end{proof}

\begin{corollary}{of Gourset's Theorem}\\
Let $f$ be holomorphic in an open set $\Omega \subseteq \mathbb{C}$ that contains a rectangle $R$ and its interior. Then
\[
\int_R f(z)dz=0
\]
\end{corollary}

\begin{proof}
    After connecting one diagonal of $R$, we can get two triangles $T_1$ and $T_2$ with
    \[
    \int_Rf(z)dz= \int_{T_1+T_2} f(z)dz= \int _{T_1} f(z)dz+ \int_{T_2}f(z)dz\xlongequal{\text{Gourset's}}0
    \]
\end{proof}

\section{Cauchy Theorem in Discs}

\noindent\textbf{Motivation:} Gourset's Thm only gives conclusion on triangular path, but we need to calculate on some more general cases. We still need to go bact to the coro of the Fundamental Thm of Calculus for Contour Integrals.

\begin{lemma}{Local Existence-of-Primitives Lemma}\\
A holomorphic function in an open disc has a primitive in that disc.
\end{lemma}

\begin{proof}
    Let $D$ be an open disc, WLOG, centered at the origin and let $f$ be holomorphic in $D$. Note that $\forall z\in D$, we have a path $\gamma_{0,z}$ consisting of two line segments, one horizontal and the other vertical. Define
    \[
    F(z):= \int_{\gamma_{0,z}} f(w)dw
    \]
    Then we claim that $F$ is holomorphic in $D$ and that $F'(z)=f(z)$. $\forall z\in D$, since $D$ is open, then $\exists r>0$, s.t.: $B_r(z) \subseteq D$. Hence, $\exists h\in \mathbb{C},$ s.t.: $z+h\in D$. Then we have:
    \begin{equation*}
        F(z+h)-F(z):= \int_{\gamma_{0,z+h}} f(w)dw- \int_{\gamma_{0,z}}f(w)dw \xlongequal{\text{Gourset's}} \int_{\eta} f(w)dw \tag{*}
    \end{equation*}
    where $\eta$ denotes the line segment from $z$ to $z+h$. By the continuity of $f$, we can write $f(w)=f(z)+ \psi(w)$ where $\psi$ denotes the error term, i.e.: $\psi(w) \to 0$ as $w\to z$. Therefore, 
    \begin{align*}
        F(z+h)- F(z) &\xlongequal{(*)} \int_{\eta} f(w)dw= f(z) \int_{\eta} dw+ \int_\eta \psi(w)dw= f(z) h+\int_\eta \psi(w)dw
    \end{align*}
    Since $|\int_\eta \psi(w)dw|\leq \sup_{w\in \eta} |\psi(w)| \cdot |h|$, then
    \[
    F'(z)= \lim_{h\to 0} \frac{F(z+h) -F(z)}{h}=f(z)
    \]
\end{proof}

\noindent\textbf{Remark:} In fact, the proof of Local Existence-of-Primitives Lemma holds not only for discs, but for the region with a good shape which allows every point $z$ in it yielding $\gamma_{0,z}$. We call a \textit{toy contour} any closed curve where the notion of interior is obvious and a construction similar to that in the proof of Local Existence-of-Primitives Lemma is possible in a neighborhood of the curve and its interior. 

Then we use an important example of toy contours to prove the Cauchy's Integral Formula later -- keyhole contours $\Gamma$, consisting of two almost complete circles, one larger and one smaller, connected by a narrow corridor.

\begin{theorem}{Cauchy's Theorem in Disc}\\
Let $f$ be a holomorphic function in a disc $D$. Then $\forall$ closed curve $\gamma$ in $D$, we have
\[
\int_\gamma f(z)dz=0
\]
\end{theorem}

\begin{proof}
    It's immediately from the Fundamental Thm of Calculus for Contour Integrals and the Local Existence-of-Primitives Lemma.
\end{proof}

\begin{corollary}{of Cauchy's Theorem in Discs}\\
If $f$ is holomorphic in an open subset of $\mathbb{C}$ containing the circle $C$ and its interior. Then
\[
\int_C f(z)dz=0
\]
\end{corollary}

\begin{proof}
    Let $D$ be the open disc with boundary $C$. Then there is a slightly larger disc $D'$ containing $D$, such that $f$ is holomorphic in $D'$. We apply Cauchy's Thm in $D'$ and then we're done.
\end{proof}

\section{Cauchy's Integral Formula}

\begin{theorem}{Cauchy's Integral Formula (CIF)}\\
Let $f$ be holomorphic in an open set that contains the closure of an open disc $D$ and let $C$ denotes the boundary circle of $D$ with the positive orientation. Then $\forall z\in D$,
\[
f(z)= \frac{1}{2\pi i} \int_C \frac{f(\zeta)}{\zeta- z} d\zeta
\]
\end{theorem}

\begin{proof}
    Consider the keyhole contour $\Gamma_{\delta,\epsilon}$ with $\delta$ the width of the corridor, $\epsilon$ the radius of the smaller circle $C_\epsilon$ at $z$ and $C$ the larger circle. Cosider the function
    \[
    F(\zeta):= \frac{f(\zeta)}{\zeta-z}
    \]
    which is holomorphic away from the point $\zeta=z$. Hence, we have
    \[
    \int_{\Gamma_{\delta,\epsilon}} F(\zeta) d\zeta=0
    \]
    by Cauchy's Thm in keyholes. Note that for $\delta\to 0$, the integral directions on the two sides of corridor are opposite and hence cancel each other out, so we have
    \[
    \int_CF(\zeta)d\zeta+ \int_{C_\epsilon} F(\zeta) d\zeta=0
    \]
    as $\delta\to 0$. Consider $F(\zeta)= \frac{f(\zeta)- f(z)}{\zeta- z}+ \frac{f(z)}{\zeta -z}$. Since $f$ is holomorphic at $z$, then $\lim_{\zeta \to z} \frac{f(\zeta)- f(z)}{\zeta -z}=f'(z)$, so the first term of $F(\zeta)$ is bounded. As for the second term,
    \[
    \int_{C_\epsilon} \frac{f(z)}{\zeta-z} d\zeta= f(z) \int_1^{2\pi} \frac{1}{\epsilon e^{-it}} (-i\epsilon e^{-it}dt)= -2\pi if(z)
    \]
    Therefore, as $\delta \to 0, \epsilon\to 0$, we have:
    \[
    0=\int_CF(\zeta) d\zeta+\lim_{\epsilon\to 0} \int_{C_\epsilon} F(\zeta) d\zeta= \int_C \frac{f(\zeta)}{\zeta-z}d\zeta- 2\pi if(z)
    \]
\end{proof}

\begin{corollary}{1 of CIF}\\
If $f$ is holomorphic in an open set $\Omega \subseteq \mathbb{C}$, then $f$ has infinitely many compex derivatives in $\Omega$. Furthermore, if $C\subseteq \Omega$ is a circle whose interior is also contained in $\Omega$, then $\forall z\in \text{int }C$,
\[
f^{(n)} (z)= \frac{n!}{2\pi i} \int_C\frac{f(\zeta)}{(\zeta- z)^{n+1}} d\zeta
\]
\end{corollary}

\begin{proof}
    It holds immediately by induction with the basic case $n=0$ being CIF and an easy computation.
\end{proof}

\begin{corollary}{2 of CIF: Cauchy Inequalities}\\
If $f$ is holomorphic in an open set that contains the closure of a disc $D$ centered at $z_0$, of radius $R$ and of boundary $C$, then
\[
|f^{(n)} (z_0)| \leq \frac{n! \cdot \sup_{z\in C} |f(z)|}{R^n}
\]
\end{corollary}

\begin{proof}
    \begin{align*}
        |f^{(n)} (z_0)|& \xlongequal{\text{coro.1}} |\frac{n!}{2\pi i} \int_C\frac{f(\zeta)}{(\zeta- z_0 )^{n+1}} d\zeta|= \frac{n!}{2\pi} |\int_0^{2\pi} \frac{f(z_0+ Re^{i\theta})}{(Re^{i\theta})^{n+1}} Rie^{i\theta} d\theta| \leq \frac{n!}{2\pi} \cdot \frac{\sup_{z\in C}|f(z)|}{R^n} \cdot 2\pi
    \end{align*}
\end{proof}

\begin{corollary}{3 of CIF: Power Series Expansion Theorem}\\
Let $f$ be holomorphic in an open set $\Omega \subseteq \mathbb{C}$. If $D$ is a disc centered at $z_0$ and whose closure is contained in $\Omega$, then $f$ has a power series expansion at $z_0$
\[
f(z)= \sum_{n=0}^\infty a_n(z-z_0)^n
\]
$\forall z\in D$, and the coefficients are given by
\[
a_n=\frac{f^{(n)}(z_0)}{n!} ,\forall n\geq 0
\]
\end{corollary}

\begin{proof}
    $\forall z\in D$, by CIF, we have:
    \[
    f(z)= \frac{1}{2\pi i} \int_C \frac{f(\zeta)}{\zeta -z} d\zeta
    \]
    where $C$ denotes the boundary of $D$. If we write
    \[
    \frac{1}{\zeta -z}= \frac{1}{\zeta-z_0- (z-z_0)}= \frac{1}{\zeta-z_0} \cdot \frac{1}{1 -\frac{z-z_0}{\zeta-z_0}}= \frac{1}{\zeta-z_0} \sum_{n=0}^\infty (\frac{z-z_0}{\zeta-z_0})^n
    \]
    then we're done.
\end{proof}

\begin{corollary}{of coro.3 of CIF / Power Series Expansion Theorem}\\
If $f$ is entire, then its power series converge everywhere in $\mathbb{C}$.
\end{corollary}

\begin{corollary}{4 of CIF: Liouville Theorem}\\
If $f$ is entire and bounded, then $f$ is constant.
\end{corollary}

\begin{proof}
    Let $|f(z)| \leq B, \forall z\in \mathbb{C}$. Then by the Cauchy Inequalities, we get: $\forall z\in \mathbb{C}, \forall R\in \mathbb{R}$,
    \[
    |f'(z)| \leq \frac{B}{R}
    \]
    Take $R\to \infty$ and then we're done.
\end{proof}

\begin{corollary}{5 of CIF: Fundamental Theorem of Algera}
\begin{enumerate}
    \item (Existence of roots). Every non-constant polynomial $P(z)= a_n z^n+\cdots +a_1z +a_0$ with complex coefficients has a root in $\mathbb{C}$.
    \item (The number of roots and factorization). Every polynomial $P(z)= a_nz^n +\cdots+ a_1z+a_0$ of degree $n\geq 1$ has precisely $n$ roots in $\mathbb{C}$. If these roots are denoted by $w_1 ,\cdots, w_n$, then $P$ can be factored as
    \[
    P(z)= a_n(z-w_1) (z-w_2) \cdots (z-w_n)
    \]
\end{enumerate}
\end{corollary}

\begin{proof}
    \begin{enumerate}
        \item Suppose on the contrary that $P$ has no roots. Assume $a_n \neq 0$.
        
        \noindent\textbf{Claim:} $1/P(z)$ is bounded.
        \begin{proof}
            \textit{of claim:} Whenever $z\neq 0$, we can write
            \[
            \frac{P(z)}{z^n}= a_n+(\frac{a_{z-1}}{z} +\cdots +\frac{a_0}{z^n})
            \]
            Since each term above in the parentheses goes to $0$ as $|z|\to \infty$, then $\exists R>0$, s.t.: whenever $|z|>R$,
            \[
            |P(z)|\geq \frac{|a_n|}{2} |z|^n
            \]
            Hence, for $|z|>R$, $|1/P(z)| \leq \frac{2}{|a_n|}\cdot \frac{1}{|z|^n} <\frac{2}{|a_n|} \cdot \frac{1}{R^n}$, so $1/P(z)$ is bounded when $|z|>R$. For $|z|\leq R$, since $P$ is continuous and has no roots in the disc $|z|\leq R$, then $P$ is bounded.
        \end{proof}
        Then by Liouville Thm, we have: $1/P(z)$ is a constant, contradiction.

        \item By 1, $P$ has a root, say $w_1$. Write $z=(z-w_1) +w_1$ and then we get
        \[
        P(z)=b_n (z-w_1)^n+ \cdots+ b_1(z-w_1) +b_0
        \]
        for some $b_n ,\cdots, b_1,b_0$ with $b_n=a_n$. Since $P(w_1)=0$, then $b_0=0$. Hence,
        \[
        P(z)= (z-w_1)(b_n(z-w_1)^{n-1} +\cdots +b_1)= (z-w_1)Q(z)
        \]
        where $Q$ is a polynomial of degree $n-1$. Then by induction, we're done.
    \end{enumerate}
\end{proof}

\section{Identity Theorem}

\noindent\textbf{Intuition:} We've seen, in the Power Series Expansion Thm, that every holomorphic function can expanded into power series. Then we must think of a question: What are the characteristics of the zeros of it?

\begin{theorem}{Identity Theorem}\\
Let $f$ be a holomorphic function in an open connected region $\Omega$ vanishes on a sequence of distinct points with a limit point in $\Omega$. Then $f$ is identically $0$ in $\Omega$.
\end{theorem}

\begin{proof}
    Let $z_0 \in \Omega$ be a limit point for the sequence $\{z_k\}_{k=1}^\infty$, such that $f(z_k)=0$. We first consider locally a disc $D\subseteq \Omega$ centered at $z_0$ and consider the power series expansion of $f$ in $D$:
    \[
    f(z)= \sum_{n=0}^\infty a_n(z-z_0)^n
    \]
    If $f$ is not identically zero in $D$, then $\exists$ a smallest integer $m$, e.t.: $a_m\neq 0$. Then we can write:
    \[
    f(z)= a_m(z-z_0)^m (1+g(z-z_0))
    \]
    where $g(z-z_0) \to 0$ as $z\to z_0$. Take $z=z_k \neq z_0$ for $k>m$. Then $a_m\neq 0$, $(z_k-z_0) ^m\neq 0$ and $1+g(z_k-z_0) \neq 0$, but $f(z_k)=0$, contradiction.

    Consider $U:= \text{int }\{z\in \Omega \mid f(z)=0\} \neq \emptyset$, which is open. On the other hand, by the continuity and the power series expansion of $f$, $U$ is closed in $\Omega$. Hence, by the connectivity of $\Omega$, the only open and closed subsets are $\emptyset$ and $\Omega$. Therefore, $U=\Omega$.
\end{proof}

\begin{corollary}{of Identity Theorem}\\
Let $f$ and $g$ be holomorphic in an open connected region $\Omega$ and let $f(z)=g(z)$ for all $z$ in some non-empty open subset of $\Omega$ (or more generally for $z$ in some distinct points with limit point in $\Omega$). Then $f= g$ throughout $\Omega$
\end{corollary}

\begin{definition}{Analytic Continuation}\\
Let $f$ and $F$ be analytic in open $\Omega$ respectively $\Omega'$, with $\Omega \subseteq \Omega'$. We say $F$ is an \textit{analytic continuation of $f$ into $\Omega'$} if $f$ and $F$ agree on $\Omega$.
\end{definition}

\section{More Applications}

\subsection{Morera's Theorem}

\begin{theorem}{Morera's Theorem}\\
Let $f$ be a continuous function in the open disc $D$ such that $\forall$ triangle $T$ contained in $D$,
\[
\int_T f(z)dz=0
\]
Then $f$ is holomorphic.
\end{theorem}

\begin{proof}
    Let $z_0$ be the center of $D$ and $\forall z_1,z_2 \in D$, let $\gamma_{z_1,z_2}$ denote the line segment from $z_1$ to $z_2$. Define
    \[
    F(z):= \int_{\gamma_{z_0,z}} f(w)dw
    \]
    Fix $z\in D$. $\forall h\in \mathbb{C}$ with $z+h \in D$, we have
    \begin{equation*}
        F(z+h)- F(z)= \int_{\gamma_{z_0,z+h}} f(w)dw- \int_{\gamma_{z_0,z}} f(w)dw \tag{*}
    \end{equation*}
    Consider a triangle $T$ with vertices $z_0,z,z+h$. By assumption,
    \begin{align*}
        0&= \int_T f(w)dw = \int_{\gamma_{z_0,z}} f(w)dw+ \int_{\gamma_{z,z+h}} f(w)dw + \int_{\gamma_{z+h, z_0}} f(w)dw\\
        &= \int_{\gamma_{z_0,z}} f(w)dw+ \int_{\gamma_{z,z+h}} f(w)dw - \int_{\gamma_{z_0, z+h}} f(w)dw \xlongequal{(*)} -(F(z+h)-F(z)) + \int_{\gamma_{z,z+h}} f(w)dw
    \end{align*}
    Hence, by def, $F'(z)= f(z)$, so we're done.
\end{proof}

\noindent\textbf{Remark:} Morera's Thm gives a converse of Cauchy's Thm.

\subsection{Sequences of Holomorphic Functions}

\noindent\textbf{Intuition:} In Real Analysis, the limit of a sequence of differentiable functions may \textit{not} differentiable. For example, by Weierstrass, every continuous function (may not differentiable) on $[0,1]$ can be approximated uniformly by polynomials.

\begin{theorem}{Holomorphic Limit Theorem}\\
If $\{f_n\}_{n=1}^\infty$ is a sequence of holomorphic functions that converges uniformly to a function $f$ in every compact subset of $\Omega$, then $f$ is holomorphic in $\Omega$.
\end{theorem}

\begin{proof}
    Let $D$ be any disc whose closure is contained in $\Omega$ and let $T$ be any triangle in $D$. By Gourset's, $\forall n$, we have
    \[
    \int_T f_n(z)dz=0
    \]
    Since $f_n\to f$ uniformly in $\overline{D}$, then $f$ is continuous and
    \[
    \int_Tf_n(z)dz \to \int_T f(z)dz
    \]
    Hence, $\int_T f(z)dz=0$. Therefore, by Morera's, we're done.
\end{proof}

\begin{theorem}{Uniform Convergence of Derivatives Theorem}\\
Under the hypothesis of the Holomorphic Limit Thm, the sequence of derivatives $\{f'_n \}_{n=1}^\infty$ converges uniformly to $f'$ on every compact subset of $\Omega$.
\end{theorem}

\begin{proof}
    WLOG, assume $\{f_n\}_{n=1}^\infty$ uniformly converges on all of $\Omega$. Define $\Omega _\delta:= \{z\in \Omega \mid \overline{D_\delta (z)} \subseteq \Omega \}$.

    \noindent\textbf{Claim:} For any holomorphic function $F$ in $\Omega$, we have:
    \[
    \sup_{z\in \Omega_\delta} |F'(z)| \leq \frac{1}{\delta} \sup_{\zeta \in \Omega} |F(\zeta) |
    \]
    \begin{proof}
        \textit{of claim:} By CIF, $F'(z)= \frac{1}{2\pi i} \int_{C_\delta (z)} \frac{F(\zeta)}{(\zeta- z)^2} d\zeta$, where $C_\delta (z)$ denotes the circle centered at $z$ and of radius $\delta$. Then by the ML Inequality, we get:
        \begin{align*}
            |F'(z)| &\leq \frac{1}{2\pi} \int_{C_\delta(z)} \frac{|F( \zeta )|}{|\zeta-z|^2} d\zeta \overset{(\text{ML})}{\leq} \frac{1}{2\pi} \sup_{\zeta \in \Omega} |F(\zeta)| \frac{1}{\delta^2} 2\pi \delta= \frac{1}{\delta} \sup_{\zeta \in \Omega} |F(\zeta)|
        \end{align*}
    \end{proof}
    Let $F=f_n-f$. Then by claim, we have:
    \[
    \sup_{z \in \Omega_\delta} |f'_n(z) -f'(z)| \leq \sup_{\zeta\in \Omega} |f_n(z)- f(\zeta)| \to 0
    \]
\end{proof}

\noindent\textbf{Remark:} We may conclude that the Uniform Convergence of Derivatives Thm can be promoted for $f^{(k)}$.


\subsection{Holomorphic Functions defined in terms of Integrals}

\noindent\textbf{Motivation:} Consider a function
\[
f(z):= \int_a^b F(z,s) ds
\]
where $F$ is holomorphic in $\Omega \subseteq \mathbb{C}$ for $z$ and continuous for $s$. This integral is easy to show that it is well-defined, but whether $f$ is holomorphic is \textit{not} trivial.

\begin{theorem}{Holomorphic Integral Theorem}\\
Let $F(z,s)$ be defined for $(z,s) \in \Omega \times [0,1]$ where $\Omega$ is an open subset of $\mathbb{C}$. Suppose $F$ satisfies the following properties:
\begin{enumerate}
    \item $F(z,s)$ is holomorphic in $z$ for each $s$.
    \item $F$ is continuous on $\Omega \times [0,1]$.
\end{enumerate}
Then the function $f$, defined on $\Omega$ by $z \mapsto \int_0^1 F(z,s)ds$, is holomorphic.
\end{theorem}

\begin{proof}
    $\forall n\geq 1$, consider $f_n(z):= (1/n) \sum_{k=1}^n F(z,k/n)$. Then by property 1, $f_n$ is holomorphic in all of $\Omega$. Let $D$ be any open disc whose closure is contained in $\Omega$. Then $\overline{D} \times [0,1]$ is compact in $\mathbb{C} \times \mathbb{R}$. Hence, by property 2, $F$ is uniformly continuous on $\overline{D} \times [0,1]$, so $\forall \epsilon>0, \exists \delta>0,$ s.t.: whenever $|s_1-s_2|<\delta$,
    \[
    \sup_{z\in D} |F(z,s_1)- F(z,s_2)|<\epsilon
    \]
    Then $\forall n \geq 1/n, \forall z\in D$, we have:
    \[
    |f_n(z)-f(z)|= |\sum_{n=1}^k \int_{\frac{k-1}{n}}^{\frac{k}{n}} (F(z,\frac{k}{n}) -F(z,s)) ds| \leq \sum_{k=1}^n \int_{\frac{k-1}{n}}^{\frac{k}{n}} |F(z,\frac{k}{n}) -F(z,s)| ds< \sum_{k=1}^n \frac{\epsilon}{n}= \epsilon
    \]
    Hence, $f_n\to f$ uniformly. Therefore, by the Holomorphic Limit Thm, we're done.
\end{proof}

\subsection{Schawarz Reflection Principle}

\noindent\textbf{Motivation:} In Real Analysis, the extension of functions is a common and easy method. But in Complex Analysis, the Identity Thm implies that the analytic continuation is difficult to complete.

\noindent\textbf{Convention:} Let $\Omega$ be an open subset of $\mathbb{C}$.
\begin{itemize}
    \item $\Omega$ is \textit{symmetric} if $z\in \Omega \Leftrightarrow \overline{z} \in \Omega$.
    \item $\Omega^+:= \{z\in \Omega \mid \text{im }z>0\}, \Omega^-:= \{ z\in \Omega \mid \text{im }z<0\}$.
    \item $I_{\Omega} := \Omega \cap \mathbb{R}$, which is the common boundary of both $\Omega^+$ and $\Omega^-$.
\end{itemize}

\begin{lemma}{Symmetry Principle}\\
Let $f^+$ and $f^-$ be holomorphic functions in $\Omega^+$ respectively $\Omega^-$, that extend continuously to $I_\Omega$ and that
\[
f^+(x)= f^-(x) ,\forall x\in I_\Omega
\]
Then $f$ defined on $\Omega$ by
\[
f(z):= \begin{cases}
    f^+(z) &\text{,if }z\in \Omega^+\\
    f^+(z)=f^-(z) &\text{,if }z\in I_\Omega\\
    f^-(z) &\text{,if }z\in \Omega^-\\
\end{cases}
\]
is holomorphic on all of $\Omega$.
\end{lemma}

\begin{proof}
    The continuity of $f$ is implied directly by assumption. Then it suffices to show that $f$ is holomorohic on $I_\Omega$: Let $D$ be any disc centered at a point in $I_\Omega$ and contained in $\Omega$. For any triangle $T$ in $D$, there are three cases:
    \begin{itemize}
        \item $T\cap I_\Omega =\emptyset$. By Gourset's, we have
        \[
        \int_T f(z)dz=0
        \]
        \item Either vertex or side of $T$ is on $I_\Omega$. Let $T_\epsilon$ denote the triangle translated $\epsilon$ into $\Omega^+$. Then by the continuity of $f$ and the conclusion in the first case, we have: as $\epsilon \to 0$,
        \[
        0 \xlongequal{\text{case 1}} \int_{T_\epsilon} f(z)dz \xrightarrow{\text{cont of }f} \int_T f(z)dz
        \]
        Hence, $\int_T f(z)dz=0$.

        \item $I_\Omega$ intersects $T$ precisely at the two points. Note that we can easily divide $T$ into some smaller triangles which satisfies the second case and applying the conclusion of the second case, we have:
        \[
        \int_T f(z)dz=0
        \]
    \end{itemize}
    Therefore, by Morera's, we're done.
\end{proof}

\begin{theorem}{Schwarz Reflection Principle}\\
Let $f$ be a holomorphic function in $\Omega^+$ such that $f$ extends continuously to $I_\Omega$ and that $f$ is real-valued on $I_\Omega$. Then there is a function $F$ holomorphic in all of $\Omega$, s.t.: $F= f$ on $\Omega^+$.
\end{theorem}

\begin{proof}
    Define $F(z):= \overline{f(z)} ,\forall z\in \Omega^-$, which is obviously holomorphic in $\Omega^-$. Since $f$ is real-valued on $I_\Omega$, then $\overline{f(x)}=f(x), \forall x\in I$. Hence, $F$ extends continuously up to $I_\Omega$. Therefore, by the Symmetry Principle, we're done.
\end{proof}

\subsection{Runge's Approximation Theorem}

\noindent\textbf{Motivation:} In Real Analysis, every continuous function on a compact interval, by Weierstrass's Thm, can be approximated uniformly by polynomials. So we ask: What conditions on a compact set $K\subseteq \mathbb{C}$ guarentee that any function holomorphic in an open neighborhood of this set can be approximated by polynomials on $K$?

\begin{theorem}{Runge's Approximation Theorem}\\
Any function holomorphic in an open neighborhood of a compact set $K$ can be approximated uniformly on $K$ by rational functions whose singularities are in $K^C$. Furthermore, if $K^C$ is connected, then any function holomorphic in an open neighborhood of $K$ can be approximated uniformly on $K$ by polynomials.
\end{theorem}

\begin{proof}
    \begin{lemma}{1: Cauchy Integral Representation Lemma on Compact Sets}\\
    Suppose $f$ is holomorphic in an open set $\Omega$ and that $K\subseteq \Omega$ is compact. Then there exist finitely many segment $\gamma_1 ,\cdots, \gamma_N$ in $\Omega \setminus K$, s.t.: $\forall z\in K,$
    \[
    f(z)=\sum_{n=1}^N \frac{1}{2\pi i} \int_{\gamma_n} \frac{f(\zeta)}{\zeta-z} d\zeta
    \]
    \end{lemma}

    \begin{proof}
        \textit{of lemma 1:} Take a grid of length $d:= \frac{1}{2\sqrt{2}} d(K,\Omega^C)$ and let $\mathcal{Q}:= \{ Q_1 ,\cdots, Q_M\}$ denote the finite collection of squares in the grid that intersect $K$. Define $\gamma_1 ,\cdots, \gamma_N$ by the sides of squares in $\mathcal{Q}$ that do not belong to two adjacent squares in $\mathcal{Q}$. Then $\gamma_n \cap K= \emptyset ,\forall n$. By the def of $d$, we have: $\gamma_n \subseteq \Omega, \forall n$. $\forall z\in K$ that is not on the boundary of a square in $\mathcal{Q}$, $\exists j$, s.t.: $z\in Q_j$. Hence, by Cauchy's,
        \[
        \frac{1}{2\pi i} \int_{\partial Q_m} \frac{f(\zeta)}{\zeta -z} d\zeta= \begin{cases}
            f(z) &\text{,if }m=j\\
            0&\text{,if }m\neq j
        \end{cases}
        \]
        Therefore, 
        \[
        f(z)= \sum_{m=1}^M \frac{1}{2\pi i} \int_{\partial Q_m} \frac{f(\zeta)}{\zeta -z} d\zeta
        \]
        Since the integral over two adjacent squares cancels, then
        \[
        f(z)=\sum_{n=1}^N \frac{1}{2\pi i} \int_{\gamma_n} \frac{f(\zeta)}{\zeta -z} d\zeta
        \]
        Finally, since $\gamma_n\subseteq K^C, \forall n$, then by the continuity of $f$, this relation can be promoted for all $z\in K$.
    \end{proof}

    \begin{lemma}{2: Rational Approximation of Contour Integrals}\\
    $\forall$ line segment $\gamma \subseteq \Omega \setminus K$, $\exists$ a sequence of rational functions with singularities on $\gamma$ that approximate the integral $\int_\gamma f(\zeta)/ (\zeta-z) d\zeta$ uniformly on $K$.
    \end{lemma}

    \begin{proof}
        \textit{of lemma 2:} Let $\gamma(t)$ be a parameterization for $\gamma$. Then
        \[
        \int_\gamma \frac{f(\zeta)}{\zeta -z} d\zeta= \int_0^1 \frac{f(\gamma(t))}{ \gamma (t)-z} \gamma'(t)dt
        \]
        Since $\gamma \cap K=\emptyset$, then the integrand $F(z,t):= \frac{f(\gamma (t))}{ \gamma(t)-z} \gamma' (t)$ is continuous on $K\times [0,1]$. Since $K$ is compact, then applying the same argument in the proof of the Holomorphic Integral Thm, we see that the Riemann sums of the integral $\int_0^1 F(z,t)dt$ approximate it uniformly on $K$. Since each of these Riemann sums is a rational function with singularities on $\gamma$, then we're done.
    \end{proof}
    By lemma 1, $f$ can be represented by finitely many contour integrals on $\gamma_n$ and then each of them, by lemma 2, can be approximated uniformly by rational functions. Hence, the first part of the Runge's Approximation Thm holds. As for the further part, we need one more lemma:

    \begin{lemma}{3: Pole Pushing Lemma}\\
    If $K^C$ is connected and $z_0 \notin K$, then the function $1/(z-z_0)$ can be approximated uniform on $K$ by polynomials.
    \end{lemma}

    \begin{proof}
        \textit{of lemma 3:} Choose a point $z_1$ that is outside a large open disc $D$ centered at the origin and containing $K$. Then
        \[
        \frac{1}{z-z_1}= -\frac{1}{z_1} \cdot \frac{1}{1-\frac{z}{z_1}}= \sum_{n=1}^\infty -\frac{z^n}{z^{n+1}_1}
        \]
        where the series converges uniformly $\forall z\in K\subseteq D$. Hence, the partial sums of this series are polynomials that approximate uniformly $1/(z-z_1)$ on $K$. Similarly, $\forall k\in \mathbb{N}_+$, $1/(z-z_k)$ can be approximated uniformly on $K$ by polynomials. Now it suffices to prove that $1/ (z-z_0)$ can be approximated uniformly on $K$ by polynomials in $1/(z-z_1)$:

        Since $K^C$ is connected and hence path-connected, then $\exists$ a path $\gamma$ fro $z_0$ to $z_1$, contained in $K^C$. Define
        \[
        \rho:= \frac{1}{2} d(K,\gamma)
        \]
        which is positive since $K$ and $\gamma$ are compact. Then choose finitely many points $z_0=w_0, w_1 ,\cdots, w_l=z_1$, s.t.: $0\leq j<l$, we have $w_j\in \gamma$ and 
        \[
        |w_j- w_{j+1}|<\rho
        \]
        Fix $w\in \gamma$. $\forall w'\in \gamma$ with $|w-w'|<\rho$, note that
        \[
        \frac{1}{z-w}= \frac{1}{z-w'} \cdot \frac{1}{1-\frac{w-w'}{z-w'}}= \sum_{n=0} ^\infty \frac{(w-w')^n}{(z-w')^{n+1}}
        \]
        Hence, $\frac{1}{z-w}$ can be approximated uniformly by $\frac{1}{z-w'}$, since $|\frac{w-w'}{z-w'}|< \frac{\rho}{2\rho} =\frac{1}{2}<1$. Therefore, we can push $\frac{1}{z-z_0}=\frac{1}{z-w_0}$ to gradually $\frac{1}{z-w_1} ,\frac{1}{z-w_2} ,\cdots,$ and finally to $\frac{1}{z-w_l}=\frac{1}{z-z_1}$.
    \end{proof}
    Then by lemma 3, we can push the singularities of rational functions all the way to infinity, thereby transforming the rational functions into polynomials.
\end{proof}
